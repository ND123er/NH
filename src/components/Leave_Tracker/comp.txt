import React, { useEffect, useState } from "react";
import {
  Box,
  Flex,
  Select,
  Text,
  Button,
  Input,
  InputGroup,
  InputLeftElement,
  InputRightElement,
  Table,
  Thead,
  Tbody,
  Tr,
  Th,
  Td,
  Modal,
  ModalOverlay,
  ModalContent,
  ModalHeader,
  ModalBody,
  ModalFooter,
  ModalCloseButton,
  FormControl,
  FormLabel,
  FormErrorMessage,
  Textarea,
  useDisclosure,
  useToast,
  Spinner,
  HStack,
  CloseButton,
  Badge,
  Menu,
  MenuButton,
  MenuList,
  Checkbox,
  VStack,
} from "@chakra-ui/react";

import { SearchIcon } from "@chakra-ui/icons";
import { FiUpload, FiPlus, FiEye, FiFilter, FiFileText } from "react-icons/fi";
import { decodeTokenPayload, getOrgCode } from "../../utils/token";
import api from "../../api/axios";

export default function CompOff() {
  const toast = useToast();
  const { isOpen, onOpen, onClose } = useDisclosure();

  const decoded = decodeTokenPayload();
  const organisation_id = getOrgCode();

  const [requests, setRequests] = useState([]);
  const [loading, setLoading] = useState(false);
  const [submitting, setSubmitting] = useState(false);

  const [form, setForm] = useState({
    worked_on: "",
    days: "",
    compensation_type: "",
    reason: "",
  });

  const [evidence, setEvidence] = useState([]);
  const [errors, setErrors] = useState({});

  // Admin-only review state
  const isAdmin = Boolean(decoded?.is_admin ?? decoded?.isAdmin);
  const [adminStatus, setAdminStatus] = useState("");
  const [adminCompensationType, setAdminCompensationType] = useState("");
  const [adminEmployeeId, setAdminEmployeeId] = useState("");
  const [adminPage, setAdminPage] = useState(1);
  const [adminLimit] = useState(20);
  const [selectedRequest, setSelectedRequest] = useState(null);
  const [adminLoading, setAdminLoading] = useState(false);
  const [adminActionLoading, setAdminActionLoading] = useState(false);
  const [adminDetailsOpen, setAdminDetailsOpen] = useState(false);
  const [approveOpen, setApproveOpen] = useState(false);
  const [rejectOpen, setRejectOpen] = useState(false);
  const [approvalForm, setApprovalForm] = useState({
    expires_on: "",
    admin_note: "",
  });
  const [rejectionReason, setRejectionReason] = useState("");
  const [adminErrors, setAdminErrors] = useState({});

  // User-side request preview
  const [userSelectedRequest, setUserSelectedRequest] = useState(null);
  const [userDetailsOpen, setUserDetailsOpen] = useState(false);
  const [evidencePreviewOpen, setEvidencePreviewOpen] = useState(false);
  const [selectedEvidence, setSelectedEvidence] = useState(null);
  const [evidencePreviewUrl, setEvidencePreviewUrl] = useState("");
  const [evidencePreviewLoading, setEvidencePreviewLoading] = useState(false);

  /* ================= GET COMP OFF REQUESTS ================= */

  const fetchCompOffRequests = async () => {
    try {
      setLoading(true);

      const res = await api.get("/leave/compoff/requests", {
        params: {
          organisation_id,
          //   ...(isAdmin
          //     ? {
          //         status: adminStatus || undefined,
          //         compensation_type: adminCompensationType || undefined,
          //         employee_id: adminEmployeeId || undefined,
          //         page: adminPage,
          //         limit: adminLimit,
          //       }
          //     : {}),
          ...(isAdmin
            ? {
                page: adminPage,
                limit: adminLimit,
                ...(adminStatus && { status: adminStatus }),
                ...(adminCompensationType && {
                  compensation_type: adminCompensationType,
                }),
                ...(adminEmployeeId && {
                  employee_id: adminEmployeeId,
                }),
              }
            : {}),
        },
      });

      const responseData = res?.data;
      const data = Array.isArray(responseData)
        ? responseData
        : Array.isArray(responseData?.data)
          ? responseData.data
          : Array.isArray(responseData?.requests)
            ? responseData.requests
            : [];

      setRequests(data);
    } catch (err) {
      toast({
        title: "Failed to load Comp Off requests",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to load Comp Off requests.",
        status: "error",
        duration: 3000,
        isClosable: true,
      });
    } finally {
      setLoading(false);
      setAdminLoading(false);
    }
  };

  useEffect(() => {
    if (!organisation_id) return;

    if (isAdmin) {
      setAdminLoading(true);
      fetchCompOffRequests();
    } else {
      fetchCompOffRequests();
    }
  }, [
    organisation_id,
    isAdmin,
    adminStatus,
    adminCompensationType,
    adminEmployeeId,
    adminPage,
  ]);

  /* ================= ADMIN REVIEW ================= */

  const openRequestDetails = async (request) => {
    try {
      setAdminActionLoading(true);

      const requestId = request?._id || request?.id;
      const res = await api.get(`/leave/compoff/requests/${requestId}`, {
        params: { organisation_id },
      });

      const responseData = res?.data;
      const details =
        responseData?.data || responseData?.request || responseData || request;

      setSelectedRequest(details);
      setAdminDetailsOpen(true);
    } catch (err) {
      toast({
        title: "Failed to load request details",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to load the request details.",
        status: "error",
        duration: 3000,
        isClosable: true,
      });
    } finally {
      setAdminActionLoading(false);
    }
  };

  const openApproveModal = () => {
    setApprovalForm({
      expires_on: "",
      admin_note: "",
    });
    setAdminErrors({});
    setApproveOpen(true);
  };

  const openRejectModal = () => {
    setRejectionReason("");
    setAdminErrors({});
    setRejectOpen(true);
  };

  const approveRequest = async () => {
    if (!selectedRequest) return;

    const compensationType = selectedRequest.compensation_type;

    if (
      compensationType === "LEAVE" &&
      approvalForm.expires_on &&
      approvalForm.expires_on < new Date().toISOString().split("T")[0]
    ) {
      setAdminErrors({
        expires_on: "Expiry date cannot be earlier than today.",
      });
      return;
    }

    try {
      setAdminActionLoading(true);

      const payload =
        compensationType === "LEAVE"
          ? {
              organisation_id,
              expires_on: approvalForm.expires_on || null,
              admin_note: approvalForm.admin_note.trim(),
            }
          : {
              organisation_id,
              admin_note: approvalForm.admin_note.trim(),
            };

      await api.put(
        `/leave/compoff/requests/${
          selectedRequest._id || selectedRequest.id
        }/approve`,
        payload,
      );

      toast({
        title: "Comp Off request approved",
        status: "success",
        duration: 3000,
        isClosable: true,
      });

      setApproveOpen(false);
      setAdminDetailsOpen(false);
      setSelectedRequest(null);
      fetchCompOffRequests();
    } catch (err) {
      toast({
        title: "Failed to approve Comp Off request",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to approve the request.",
        status: "error",
        duration: 4000,
        isClosable: true,
      });
    } finally {
      setAdminActionLoading(false);
    }
  };

  const rejectRequest = async () => {
    if (!selectedRequest) return;

    if (!rejectionReason.trim()) {
      setAdminErrors({
        rejection_reason: "Rejection reason is required.",
      });
      return;
    }

    if (rejectionReason.trim().length > 2000) {
      setAdminErrors({
        rejection_reason: "Rejection reason cannot exceed 2,000 characters.",
      });
      return;
    }

    try {
      setAdminActionLoading(true);

      await api.put(
        `/leave/compoff/requests/${
          selectedRequest._id || selectedRequest.id
        }/reject`,
        {
          organisation_id,
          rejection_reason: rejectionReason.trim(),
        },
      );

      toast({
        title: "Comp Off request rejected",
        status: "success",
        duration: 3000,
        isClosable: true,
      });

      setRejectOpen(false);
      setAdminDetailsOpen(false);
      setSelectedRequest(null);
      fetchCompOffRequests();
    } catch (err) {
      toast({
        title: "Failed to reject Comp Off request",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to reject the request.",
        status: "error",
        duration: 4000,
        isClosable: true,
      });
    } finally {
      setAdminActionLoading(false);
    }
  };

  const openUserRequestDetails = async (request) => {
    try {
      setLoading(true);

      const requestId = request?._id || request?.id;
      const res = await api.get(`/leave/compoff/requests/${requestId}`, {
        params: { organisation_id },
      });

      const responseData = res?.data;
      const details =
        responseData?.data ||
        responseData?.request ||
        responseData ||
        request;

      setUserSelectedRequest(details);
      setUserDetailsOpen(true);
    } catch (err) {
      toast({
        title: "Failed to load request details",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to load the request details.",
        status: "error",
        duration: 3000,
        isClosable: true,
      });
    } finally {
      setLoading(false);
    }
  };

  const closeUserDetails = () => {
    setUserDetailsOpen(false);
    setUserSelectedRequest(null);
  };

  const previewEvidence = async (file) => {
    try {
      setSelectedEvidence(file);
      setEvidencePreviewLoading(true);
      setEvidencePreviewOpen(true);

      const evidencePath = file.url?.replace(/^\/api\/v1\/hrm/, "");

      const res = await api.get(evidencePath, {
        responseType: "blob",
      });

      const blobUrl = URL.createObjectURL(res.data);
      setEvidencePreviewUrl(blobUrl);
    } catch (err) {
      setEvidencePreviewUrl("");
      toast({
        title: "Unable to preview evidence",
        description:
          err?.response?.data?.message ||
          "The evidence file could not be opened.",
        status: "error",
        duration: 3000,
        isClosable: true,
      });
    } finally {
      setEvidencePreviewLoading(false);
    }
  };

  const closeEvidencePreview = () => {
    if (evidencePreviewUrl) {
      URL.revokeObjectURL(evidencePreviewUrl);
    }
    setEvidencePreviewUrl("");
    setSelectedEvidence(null);
    setEvidencePreviewOpen(false);
  };

  /* ================= FORM ================= */

  const handleChange = (e) => {
    const { name, value } = e.target;

    setForm((prev) => ({
      ...prev,
      [name]: value,
    }));

    setErrors((prev) => ({
      ...prev,
      [name]: "",
    }));
  };

  const handleEvidenceChange = (e) => {
    const files = Array.from(e.target.files || []);

    if (files.length > 5) {
      setErrors((prev) => ({
        ...prev,
        evidence: "You can upload a maximum of 5 files.",
      }));
      return;
    }

    const invalidFile = files.find((file) => {
      const allowedTypes = [
        "image/jpeg",
        "image/png",
        "image/webp",
        "application/pdf",
      ];

      return !allowedTypes.includes(file.type) || file.size > 5 * 1024 * 1024;
    });

    if (invalidFile) {
      setErrors((prev) => ({
        ...prev,
        evidence: "Each file must be JPEG, PNG, WEBP or PDF and maximum 5MB.",
      }));
      return;
    }

    setEvidence(files);

    setErrors((prev) => ({
      ...prev,
      evidence: "",
    }));
  };

  const validateForm = () => {
    const newErrors = {};

    if (!form.worked_on) {
      newErrors.worked_on = "Please select the worked date.";
    }

    if (!form.days) {
      newErrors.days = "Please select the number of days.";
    }

    if (!form.compensation_type) {
      newErrors.compensation_type = "Please select compensation type.";
    }

    if (!form.reason.trim()) {
      newErrors.reason = "Reason is required.";
    }

    if (evidence.length < 1) {
      newErrors.evidence = "Please upload at least one evidence file.";
    }

    setErrors(newErrors);

    return Object.keys(newErrors).length === 0;
  };

  /* ================= SUBMIT ================= */

  const handleSubmit = async () => {
    if (!validateForm()) return;

    try {
      setSubmitting(true);

      const formData = new FormData();

      formData.append("organisation_id", organisation_id);
      formData.append("worked_on", form.worked_on);
      formData.append("days", form.days);
      formData.append("compensation_type", form.compensation_type);
      formData.append("reason", form.reason.trim());

      //   evidence.forEach((file) => {
      //     formData.append("evidence", file);
      //   });
      evidence.forEach((file) => {
        formData.append("evidence", file);
      });
      //   await api.post("/leave/compoff/requests", formData);
      await api.post("/leave/compoff/requests", formData, {
        headers: {
          "Content-Type": "multipart/form-data",
        },
      });
      toast({
        title: "Comp Off request submitted",
        status: "success",
        duration: 3000,
        isClosable: true,
      });

      setForm({
        worked_on: "",
        days: "",
        compensation_type: "",
        reason: "",
      });

      setEvidence([]);
      setErrors({});

      onClose();

      fetchCompOffRequests();
    } catch (err) {
      toast({
        title: "Failed to submit Comp Off request",
        description:
          err?.response?.data?.message ||
          err?.response?.data?.msg ||
          "Unable to submit your request.",
        status: "error",
        duration: 4000,
        isClosable: true,
      });
    } finally {
      setSubmitting(false);
    }
  };

  const closeModal = () => {
    setForm({
      worked_on: "",
      days: "",
      compensation_type: "",
      reason: "",
    });
    setEvidence([]);
    setErrors({});
    onClose();
  };

 
//newer
if (isAdmin) {
  const adminStatusOptions = [
    { id: "PENDING", label: "Pending" },
    { id: "APPROVED", label: "Approved" },
    { id: "REJECTED", label: "Rejected" },
  ];

  const compensationOptions = [
    { id: "LEAVE", label: "Leave" },
    { id: "ENCASHMENT", label: "Encashment" },
  ];

  return (
    <Box
      fontFamily="Poppins"
      h="100vh"
      overflowY="auto"
      bg="#F6F7FB"
    >
      <Box
        bg="white"
        border="1px solid #cccccc"
        borderRadius="14px"
        shadow="sm"
        overflow="hidden"
      >
        {/* ================= HEADER ================= */}
        <Box py={3} pb={2}>
          <Flex
            justify="space-between"
            align="center"
            gap={3}
            wrap="wrap"
            ml={3}
            mr={3}
            mt={1}
            mb={1}
          >
            <Text
              fontSize="22px"
              fontWeight="700"
              color="#000000"
              ml={4}
            >
              Comp Off Requests
            </Text>

            {/* ================= SEARCH + FILTER ================= */}
            <Flex align="center" gap={2}>
              {/* Search */}
              <Box
                position="relative"
                w={{
                  base: "200px",
                  md: "220px",
                  lg: "320px",
                }}
                overflow="hidden"
              >
                <InputGroup size="sm">
                  <InputLeftElement pointerEvents="none">
                    <SearchIcon color="#757575" mt={2} />
                  </InputLeftElement>

                  {!adminEmployeeId && (
                    <Text
                      position="absolute"
                      left="30px"
                      top="50%"
                      transform="translateY(-50%)"
                      fontSize={{
                        base: "12px",
                        sm: "12px",
                        md: "11px",
                        lg: "14px",
                      }}
                      fontFamily="Poppins"
                      fontWeight="400"
                      color="#757575"
                      pointerEvents="none"
                      whiteSpace="nowrap"
                      zIndex={1}
                    >
                      Search by Employee ID
                    </Text>
                  )}

                  <Input
                    value={adminEmployeeId}
                    onChange={(e) => {
                      setAdminEmployeeId(e.target.value);
                      setAdminPage(1);
                    }}
                    fontFamily="Poppins"
                    h="39px"
                    borderWidth="1.5px"
                    borderColor="#cccccc"
                    borderRadius="10px"
                    pl="30px"
                    _focus={{
                      borderColor: "#cccccc",
                      boxShadow: "none",
                    }}
                  />

                  {adminEmployeeId && (
                    <InputRightElement>
                      <CloseButton
                        size="sm"
                        mt={1}
                        onClick={() => {
                          setAdminEmployeeId("");
                          setAdminPage(1);
                        }}
                      />
                    </InputRightElement>
                  )}
                </InputGroup>
              </Box>

              {/* ================= COMPENSATION FILTER ================= */}
              <Menu closeOnSelect={false}>
                <MenuButton
                  as={Button}
                  h="35px"
                  bg="white"
                  fontFamily="Poppins"
                  fontSize="14px"
                  color="#757575"
                  borderWidth="1.5px"
                  borderColor="#cccccc"
                  borderRadius="8px"
                  fontWeight={400}
                  leftIcon={<FiFilter />}
                  _hover={{
                    bg: "white",
                    borderColor: "#bdbdbd",
                  }}
                  _active={{
                    bg: "white",
                  }}
                  _focus={{
                    boxShadow: "none",
                  }}
                >
                  Filter
                </MenuButton>

                <MenuList
                  minW="210px"
                  p={3}
                  borderRadius="10px"
                  fontFamily="Poppins"
                >
                  <Text
                    fontSize="14px"
                    fontWeight="600"
                    mb={3}
                    color="#111827"
                  >
                    Compensation Type
                  </Text>

                  <VStack align="start" spacing={3}>
                    {compensationOptions.map((item) => (
                      <Checkbox
                        key={item.id}
                        isChecked={
                          adminCompensationType === item.id
                        }
                        onChange={() => {
                          setAdminCompensationType(
                            adminCompensationType === item.id
                              ? ""
                              : item.id
                          );
                          setAdminPage(1);
                        }}
                        sx={{
                          ".chakra-checkbox__control": {
                            borderColor: "#cccccc",
                          },
                          ".chakra-checkbox__control[data-checked]": {
                            bg: "#274066",
                            borderColor: "#274066",
                          },
                        }}
                      >
                        <Text fontSize="14px">
                          {item.label}
                        </Text>
                      </Checkbox>
                    ))}
                  </VStack>

                  {adminCompensationType && (
                    <Button
                      mt={4}
                      size="sm"
                      variant="ghost"
                      fontFamily="Poppins"
                      onClick={() => {
                        setAdminCompensationType("");
                        setAdminPage(1);
                      }}
                    >
                      Clear
                    </Button>
                  )}
                </MenuList>
              </Menu>
            </Flex>
          </Flex>
        </Box>

        {/* ================= DOTTED SEPARATOR ================= */}
        <Box
          w="100%"
          h="1.5px"
          bgImage="
            repeating-linear-gradient(
              to right,
              #C7C7C7 0 6px,
              transparent 6px 14px
            )
          "
          mb={4}
        />

        {/* ================= STATUS FILTER ================= */}
        <Flex
          gap={3}
          align="center"
          wrap="wrap"
          m={3}
          mt={0}
          mb={3}
        >
          <Text
            fontWeight="600"
            minW="120px"
            ml={4}
            fontFamily="Poppins"
          >
            Status
          </Text>

          <HStack spacing={2} wrap="wrap">
            {adminStatusOptions.map((s) => {
              const active = adminStatus === s.id;

              return (
                <Button
                  key={s.id}
                  size="sm"
                  onClick={() => {
                    setAdminStatus(active ? "" : s.id);
                    setAdminPage(1);
                  }}
                  borderRadius="full"
                  px={3}
                  py={1.2}
                  position="relative"
                  _focus={{ boxShadow: "none" }}
                  display="inline-flex"
                  alignItems="center"
                  gap={2}
                  borderWidth="1px"
                  borderColor={active ? "#000" : "#C7C7C7"}
                  bg="#fff"
                  color={active ? "#000" : "#C7C7C7"}
                  boxShadow={active ? "sm" : "none"}
                  height="38px"
                >
                  <Text
                    as="span"
                    fontSize="sm"
                    fontWeight={500}
                    fontFamily="Poppins"
                  >
                    {s.label}
                  </Text>

                  {active && (
                    <CloseButton
                      position="absolute"
                      top="-10px"
                      right="4px"
                      boxSize="20px"
                      minW="20px"
                      h="20px"
                      p="0"
                      fontSize="9px"
                      bg="#E5F4EC"
                      color="#000"
                      borderRadius="full"
                      onClick={(e) => {
                        e.stopPropagation();
                        setAdminStatus("");
                        setAdminPage(1);
                      }}
                      aria-label={`Clear ${s.label}`}
                      _hover={{
                        bg: "#C6EAD8",
                      }}
                    />
                  )}
                </Button>
              );
            })}
          </HStack>

          {adminStatus && (
            <Button
              size="sm"
              variant="ghost"
              onClick={() => {
                setAdminStatus("");
                setAdminPage(1);
              }}
              ml={2}
            >
              Clear
            </Button>
          )}
        </Flex>

        {/* ================= DOTTED SEPARATOR ================= */}
        <Box
          w="100%"
          h="1.5px"
          bgImage="
            repeating-linear-gradient(
              to right,
              #C7C7C7 0 6px,
              transparent 6px 14px
            )
          "
          mb={4}
        />

        {/* ================= TABLE ================= */}
        {loading || adminLoading ? (
          <Flex
            justify="center"
            align="center"
            minH="200px"
          >
            <Spinner />
          </Flex>
        ) : (
          <Box
            overflowX="auto"
            px={{ base: 2, sm: 2, md: 2, lg: 0 }}
          >
            <Table
              variant="striped"
              size="sm"
              sx={{
                "& tbody tr:nth-of-type(odd) td": {
                  background: "#f0f2f5",
                },
                "& tbody tr:nth-of-type(even) td": {
                  background: "white",
                },
                "& td, & th": {
                  border: "none !important",
                },
                "& tr": {
                  border: "none !important",
                },
                "& thead th": {
                  paddingBottom: "14px",
                },
                "& tbody tr:first-of-type td": {
                  paddingTop: "12px",
                },
              }}
            >
              <Thead
                sx={{
                  th: {
                    fontSize: "14px",
                    textTransform: "none",
                    fontWeight: "600",
                    fontFamily: "'Poppins', sans-serif",
                    color: "#000000",
                  },
                }}
              >
                <Tr>
                  <Th>Employee</Th>
                  <Th>Worked On</Th>
                  <Th>Days</Th>
                  <Th>Compensation Type</Th>
                  <Th>Reason</Th>
                  <Th>Evidence</Th>
                  <Th>Status</Th>
                  <Th>Action</Th>
                </Tr>
              </Thead>

              <Tbody>
                {requests.length === 0 ? (
                  <Tr>
                    <Td colSpan={8}>
                      <Text
                        textAlign="center"
                        py={8}
                        color="gray.500"
                        fontFamily="Poppins"
                      >
                        No Comp Off requests found
                      </Text>
                    </Td>
                  </Tr>
                ) : (
                  requests.map((item) => (
                    <Tr key={item._id || item.id}>
                      <Td>
                        {item.employee_name ||
                          item.employee_id ||
                          "-"}
                      </Td>

                      <Td>
                        {item.worked_on || "-"}
                      </Td>

                      <Td>
                        {item.days ?? "-"}
                      </Td>

                      <Td>
                        {item.compensation_type || "-"}
                      </Td>

                      <Td maxW="280px">
                        {item.reason || "-"}
                      </Td>

                      <Td>
                        {item.evidence?.length > 0 ? (
                          <Button
                            size="sm"
                            variant="ghost"
                            leftIcon={<FiFileText />}
                            onClick={() => previewEvidence(item.evidence[0])}
                            fontFamily="Poppins"
                            color="#274167"
                            _hover={{
                              bg: "#F3F6FA",
                            }}
                          >
                            {item.evidence.length > 1
                              ? `${item.evidence.length} Files`
                              : "Evidence"}
                          </Button>
                        ) : (
                          <Text color="gray.400">-</Text>
                        )}

            <Box display="flex" flexDirection="column" gap={4}>
              {/* Worked On */}
              <FormControl isRequired isInvalid={!!errors.worked_on}>
                <FormLabel>Worked On</FormLabel>

                <Input
                  type="date"
                  name="worked_on"
                  value={form.worked_on}
                  max={new Date().toISOString().split("T")[0]}
                  onChange={handleChange}
                />

                <FormErrorMessage>{errors.worked_on}</FormErrorMessage>
              </FormControl>

              {/* Days */}
              <FormControl isRequired isInvalid={!!errors.days}>
                <FormLabel>Days</FormLabel>

                <Select
                  name="days"
                  value={form.days}
                  placeholder="Select days"
                  onChange={handleChange}
                >
                  <option value="0.5">0.5</option>
                  <option value="1">1</option>
                </Select>

                <FormErrorMessage>{errors.days}</FormErrorMessage>
              </FormControl>

              {/* Compensation Type */}
              <FormControl isRequired isInvalid={!!errors.compensation_type}>
                <FormLabel>Compensation Type</FormLabel>

                <Select
                  name="compensation_type"
                  value={form.compensation_type}
                  placeholder="Select compensation type"
                  onChange={handleChange}
                >
                  <option value="LEAVE">Leave</option>

                  <option value="ENCASHMENT">Encashment</option>
                </Select>

                <FormErrorMessage>{errors.compensation_type}</FormErrorMessage>
              </FormControl>

              {/* Reason */}
              <FormControl isRequired isInvalid={!!errors.reason}>
                <FormLabel>Reason</FormLabel>

                <Textarea
                  name="reason"
                  value={form.reason}
                  placeholder="Enter reason"
                  onChange={handleChange}
                />

                <FormErrorMessage>{errors.reason}</FormErrorMessage>
              </FormControl>

              {/* Evidence */}
              <FormControl isRequired isInvalid={!!errors.evidence}>
                <FormLabel>Evidence</FormLabel>

                <Button
                  as="label"
                  htmlFor="compoff-evidence"
                  variant="outline"
                  leftIcon={<FiUpload />}
                  cursor="pointer"
                  w="fit-content"
                >
                  Upload Evidence
                </Button>

                <Input
                  id="compoff-evidence"
                  type="file"
                  display="none"
                  multiple
                  accept=".jpg,.jpeg,.png,.webp,.pdf"
                  onChange={handleEvidenceChange}
                />

                {evidence.length > 0 && (
                  <Box mt={2}>
                    {evidence.map((file) => (
                      <Text key={file.name} fontSize="sm" color="gray.600">
                        {file.name}
                      </Text>
                    ))}
                  </Box>
                )}

                <FormErrorMessage>{errors.evidence}</FormErrorMessage>

                <Text fontSize="12px" color="gray.500" mt={1}>
                  Maximum 5 files, 5MB each.
                </Text>
              </FormControl>
            </Box>
          </ModalBody>

          <ModalFooter>
            <Button variant="ghost" mr={3} onClick={closeModal}>
              Cancel
            </Button>

            <Button
              bg="#274167"
              color="white"
              _hover={{ bg: "#1f3143" }}
              onClick={handleSubmit}
              isLoading={submitting}
            >
              Submit
            </Button>
          </ModalFooter>
        </ModalContent>
      </Modal>
    </Box>
  );
}
